2    if lo == hi                  // one element: then i == j == lo
3        return A[lo]
4    mid = ⌊(lo + hi) / 2⌋
5    if j ≤ mid                  // case 1: all in the left half
6        return RANGE-MIN(A, lo, mid, i, j)
7    elseif i > mid               // case 2: all in the right half
8        return RANGE-MIN(A, mid + 1, hi, i, j)
9    else                         // case 3: split between the halves
10        left = RANGE-MIN(A, lo, mid, i, mid)
11        right = RANGE-MIN(A, mid + 1, hi, mid + 1, j)
12        return min(left, right)
